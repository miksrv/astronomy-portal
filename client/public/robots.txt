# Admin pages and forms — not for public indexing. Pages that people reach from emails
# and messengers (/auth, /profile, /unsubscribe, /stargazing/checkin, /stargazing/payment)
# are NOT blocked here on purpose: they carry <meta name="robots" content="noindex">, and a
# crawler blocked by robots.txt never sees it - it may index the bare URL from the links.

Disallow: /stargazing/form

# Yandex-specific: strip tracking/UTM params before indexing so links shared
# with campaign/referrer tags (e.g. ?trk=..., ?ref=..., ?_ym_status-check=...)
# are treated as the same URL as the clean page instead of being indexed as
# separate duplicates with an identical title.
Clean-param: _ym_status-check&_ym_lang /
Clean-param: trk /
Clean-param: ref /
Clean-param: utm_source&utm_medium&utm_campaign&utm_term&utm_content /
